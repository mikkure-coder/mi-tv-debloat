# =============================================================================
# СПИСОК ПАКЕТОВ ПРИСТАВКИ Xiaomi TV Box S 2 ДЛЯ ОТКЛЮЧЕНИЯ
# (pm disable-user --user 0; читает ./mibox-s2/scripts/02-debloat.sh)
# =============================================================================
#
# Как читать файл:
#   - строка с именем пакета — будет отключён;
#   - строка, начинающаяся с «#», — пропускается (так пакет оставляют в покое);
#   - pm disable-user НЕ удаляет пакет, а «замораживает». Вернуть всё —
#     ./mibox-s2/scripts/99-restore.sh, вернуть один — pm enable <имя>.
#
# ОТКУДА СПИСОК: CleanYourTV, examples/xiaomi-mi-box-s-2nd-gen — там каждую
# строку отключали на этой же модели (MiTV-AFKR0, jaws) и проверяли
# перезагрузкой. НО там была прошивка Google TV 11 (сборка RTT0…), а у нас
# Android 14 (UKG3.250107.001, user/release-keys).
#
# ИМЕНА СВЕРЕНЫ С ЖИВОЙ ПРИСТАВКОЙ 2026-10-25 (pm list packages): каждое имя
# ниже на ней есть. Пометка «не проверено» значит: что сломается или не
# сломается после отключения на нашей прошивке, ещё не проверяли. Её снимаем
# по факту, после перезагрузки, как у телевизора.
# Убрано при сверке: android.autoinstalls.config.xioami.mibox3 и
# com.google.android.videos — на приставке их нет.
#
# Порядок работы: сначала DRY_RUN=1, потом группы 1–2, перезагрузка, пара дней
# жизни — и только потом остальное.
# =============================================================================

com.google.android.feedback            # не проверено: отчёты об ошибках в Google
com.google.android.partnersetup        # не проверено: «отметки» для партнёров Google (RLZ)

com.mitv.tvhome.oemtab                 # не проверено: рекламный ряд Xiaomi
com.mitv.tvhome.michannel              # не проверено: ряды контента Mi
com.mitv.tvhome.mitvplus               # не проверено: Mi TV+ (каналы с рекламой)
com.mitv.tvhome.atv                    # не проверено: PatchWall
com.xm.webcontent                      # не проверено: веб-промо Xiaomi

#android.autoinstalls.config.soc.amlogic   # не проверено


# ---- 3. ПРЕДУСТАНОВЛЕННЫЕ ПРИЛОЖЕНИЯ — по желанию владельца ---------------
# Ничего системного, но пользуешься ли ты ими — решать тебе.
# Prime Video (com.amazon.amazonvideo.livingroom) и Netflix (com.netflix.ninja)
# на приставке уже были отключены до нас — их 99 не включит.
# НЕ ДЕРЖИТСЯ (2026-10-25): после перезагрузки снова включён. Включил Google
# Play (lastDisabledCaller: com.android.vending) — он считает Play Игры своей

#com.google.android.play.games         # Play Игры
# НЕ ДЕРЖИТСЯ (2026-10-25): после перезагрузки состояние сброшено на
# «по умолчанию» (enabled=0). Кто сбрасывает — по журналу не видно.
#com.android.printspooler              # печать


# ---- 4. ОБНОВЛЕНИЯ ПРОШИВКИ — по желанию владельца -------------------------
# Пока включено, Xiaomi может прислать новую прошивку, и она вернёт отключённое.
# Выключено — приставка перестанет сама искать обновления.
# Решение владельца 2026-10-25: оставить включённым ради исправлений безопасности.
#com.xiaomi.mitv.updateservice         # обновление прошивки Xiaomi


# ---- 5. ДОМАШНИЙ ЭКРАН GOOGLE TV — ТОЛЬКО ПОСЛЕ НОВОГО ЛАУНЧЕРА -------------
# Раскомментировать, только когда лаунчер CleanYourTV поставлен через 03
# и открылся ручным запуском. 02 сам откажется, если другого дома нет.
# Что пропадёт вместе с ним: кнопки Netflix, YouTube и «Приложения» на пульте
# перестанут работать (их обрабатывал этот лаунчер) — у CleanYourTV на этой
# модели их возвращает сам новый лаунчер (см. CLAUDE.md подпроекта).
# На этой модели одного его отключения МАЛО: дом переходит к мастеру
# настройки, поэтому следом отключается и он (решение владельца 2026-10-25,
# разрешение в core-exceptions.txt). Порядок строк важен: сначала лаунчер.
# 02 не тронет мастер, пока сеть Wi-Fi принадлежит ему.
com.google.android.apps.tv.launcherx   # не проверено: домашний экран Google TV
com.google.android.tungsten.setupwraith   # не проверено: мастер первой настройки
