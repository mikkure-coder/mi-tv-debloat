# =============================================================================
# ОТДЕЛЬНЫЕ ЧАСТИ ПРИЛОЖЕНИЙ, КОТОРЫЕ ОТКЛЮЧАЕМ (05-components.sh)
# =============================================================================
#
# Чем это отличается от debloat.txt:
#   - отключается НЕ приложение целиком, а одна его часть (компонент),
#     например получатель сигнала «ТВ загрузился». Иконка остаётся,
#     приложение открывается как обычно;
#   - команда: pm disable --user 0 <пакет>/<компонент>.
#
# Откат: ./scripts/99-restore.sh — вернёт каждой части исходное состояние
#        из backup/components-before.txt (обычно `pm default-state`).
#
# Формат: «<пакет>/<полное имя компонента>», «#» — комментарий.
# Пакеты из keep.txt и ядро Android сюда ставить нельзя — 05 откажется.
# =============================================================================


# МТС ТВ стартует при каждой загрузке по сигналу BOOT_COMPLETED, который
# ловит этот получатель из push-модуля МТС (logcat 2026-10-25: «Start proc …
# for broadcast …BootCompletedReceiver»). Запрет фона (background.txt) это не
# останавливает. Без него МТС, видимо, перестанет присылать push-уведомления.
# У МТС есть второй получатель того же сигнала —
# androidx.work.impl.background.systemalarm.RescheduleReceiver (WorkManager);
# его пока не трогаем: сначала замер с одним.
ru.mts.mtstv/ru.mts.push.repeater.domain.receivers.BootCompletedReceiver   # не проверено
